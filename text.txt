  <script>
    // ---- Customize quickly (or use ?from= & ?to= in the URL) ----
    const params = new URLSearchParams(location.search);
    const from = params.get("from") || "Amir";
    const to   = params.get("to")   || "Babe";

    const fromEl = document.getElementById("fromName");
    const toEl   = document.getElementById("toName");
    const signEl = document.getElementById("sign");
    fromEl.textContent = "From: " + from;
    toEl.textContent   = "To: " + to;
    signEl.textContent = "— " + from;

    // ---- No button runs away ----
    const noBtn = document.getElementById("noBtn");
    const card  = document.getElementById("card");
    const phrases = ["Are you sure? 😭","Think again 🥺","Pls pls 🥹","Don’t do me like that 💔","Ok last chance 😤"];
    let noCount = 0;

    function moveNo(){
      const r = card.getBoundingClientRect();
      const b = noBtn.getBoundingClientRect();
      const pad = 12;
      const maxX = r.width - b.width - pad*2;
      const maxY = r.height - b.height - pad*2;

      const x = pad + Math.random() * Math.max(0, maxX);
      const y = pad + Math.random() * Math.max(0, maxY);

      noBtn.style.position = "absolute";
      noBtn.style.left = x + "px";
      noBtn.style.top  = y + "px";
      noBtn.textContent = phrases[Math.min(noCount, phrases.length-1)];
      noCount++;
    }
    noBtn.addEventListener("mouseenter", moveNo);
    noBtn.addEventListener("click", moveNo);

    // ---- Yes -> show letter + confetti ----
    const yesBtn = document.getElementById("yesBtn");
    const result = document.getElementById("result");
    const canvas = document.getElementById("confetti");
    const ctx = canvas.getContext("2d");

    function resize(){
      canvas.width = innerWidth * devicePixelRatio;
      canvas.height = innerHeight * devicePixelRatio;
      ctx.scale(devicePixelRatio, devicePixelRatio);
    }

    let pieces = [];
    function launchConfetti(){
      canvas.classList.remove("hidden");
      pieces = Array.from({length: 180}, () => ({
        x: Math.random()*innerWidth,
        y: -20 - Math.random()*innerHeight*0.3,
        w: 6 + Math.random()*6,
        h: 8 + Math.random()*10,
        vx: -2 + Math.random()*4,
        vy: 2 + Math.random()*5,
        r: Math.random()*Math.PI,
        vr: -0.2 + Math.random()*0.4
      }));

      const start = performance.now();
      function tick(t){
        ctx.clearRect(0,0,innerWidth,innerHeight);
        for(const p of pieces){
          p.x += p.vx; p.y += p.vy; p.r += p.vr;
          p.vy += 0.03;
          ctx.save();
          ctx.translate(p.x, p.y);
          ctx.rotate(p.r);
          ctx.fillRect(-p.w/2, -p.h/2, p.w, p.h);
          ctx.restore();
        }
        // stop after ~4s
        if(t - start < 4200) requestAnimationFrame(tick);
        else canvas.classList.add("hidden");
      }
      requestAnimationFrame(tick);
    }

    yesBtn.addEventListener("click", () => {
      result.classList.remove("hidden");
      yesBtn.disabled = true;
      noBtn.disabled = true;
      launchConfetti();
    });

    addEventListener("resize", () => location.reload());
    // hearts in background
    const hearts = ["💗","💖","💘","💕","💞","💓"];
    for(let i=0;i<18;i++){
      const s = document.createElement("div");
      s.className = "heart";
      s.textContent = hearts[Math.floor(Math.random()*hearts.length)];
      s.style.setProperty("--x", Math.random()*100 + "vw");
      s.style.setProperty("--s", (18 + Math.random()*22) + "px");
      s.style.setProperty("--d", (7 + Math.random()*8) + "s");
      s.style.animationDelay = (-Math.random()*10) + "s";
      document.body.appendChild(s);
    }

    // share / edit
    document.getElementById("shareBtn").addEventListener("click", async () => {
      const url = location.origin + location.pathname + `?from=${encodeURIComponent(from)}&to=${encodeURIComponent(to)}`;
      await navigator.clipboard.writeText(url);
      alert("Link copied!");
    });

    document.getElementById("editBtn").addEventListener("click", () => {
      const nf = prompt("From name:", from) || from;
      const nt = prompt("To name:", to) || to;
      const url = location.pathname + `?from=${encodeURIComponent(nf)}&to=${encodeURIComponent(nt)}`;
      location.href = url;
    });
  </script>
